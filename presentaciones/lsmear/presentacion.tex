\begin{frame}{El problema es elegir bien qué variable dividir}
\small
\begin{itemize}
  \item Una mala elección de variable puede disparar el árbol de búsqueda.
  \item La bisectriz afecta \textbf{cota inferior}, factibilidad y número de cajas.
  \item \texttt{round-robin} ignora el sistema; \texttt{largest-first} mira solo el ancho.
\end{itemize}

\begin{alertblock}{Idea clave}
\small
La variable a dividir debe reflejar el impacto real en el problema, no solo el dominio.
\end{alertblock}
\end{frame}

\begin{frame}{Las heurísticas clásicas no distinguen restricciones activas}
\small
\begin{columns}[T]
\column{0.50\textwidth}
\begin{block}{\texttt{smear} trata todas las restricciones igual}
\begin{itemize}
  \item \texttt{smearsum}, \texttt{smearmax} y \texttt{smearsumrel} agregan impactos sin ponderación.
  \item No distinguen restricciones activas e inactivas.
\end{itemize}
\end{block}

\column{0.50\textwidth}
\begin{alertblock}{\texttt{ViolationTransfer} es más intrusiva}
\begin{itemize}
  \item Usa relajación convexa y la función de Lagrange.
  \item Requiere un problema reformulado para construir $x_i^v$.
\end{itemize}
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{La idea es ponderar con multiplicadores duales óptimos}
\small
\begin{block}{Contexto y método}
\begin{itemize}
\item \textbf{lsmear} reusa una función smear tipo Lagrangiano para ponderar restricciones.
\item Linealiza el problema alrededor del punto medio de la caja.
\item Resuelve un programa lineal para estimar \emph{optimal Lagrange multipliers}.
\item Reemplaza los pesos del Lagrangiano por esa estimación dual.
\end{itemize}
\end{block}

\begin{exampleblock}{Esquema de la aproximación}
\[
g_j(\mathbf{x}) \approx g_j(\mathrm{mid}(\mathbf{x})) + \mathrm{mid}(J_{j:}) \, (\mathbf{x}-\mathrm{mid}(\mathbf{x}))
\]
La expansión de primer orden usa \(\mathrm{mid}(J_{ji})\) como sobreestimación intervalar.
\end{exampleblock}
\end{frame}

\begin{frame}{lsmear computa el impacto con un Lagrangiano}
\small
\begin{equation*}
L(\mathbf{x},\lambda^*) = f(\mathbf{x}) + \sum_{j=1}^{m} \lambda_j^*\, g_j(\mathbf{x})
\end{equation*}
\begin{itemize}
  \item $L$ pondera cada restricción con \textbf{multiplicadores duales óptimos}.
  \item $\lambda_j^*=0$ para restricciones inactivas.
\end{itemize}

\begin{equation*}
\mathrm{impacto}_i = \left|D_i\right|\,\mathrm{wid}(x_i),
\qquad
D_i = \lambda_i^* + \sum_{j=1}^{m} \lambda_{n+j}^*\, J_{ji}
\end{equation*}
\begin{itemize}
  \item $D_i$ agrega el efecto de la variable $x_i$ en la linealización.
  \item \alert{Se bisecciona la variable con mayor impacto.}
\end{itemize}
\end{frame}

\begin{frame}{\texttt{lsmear} linealiza y luego selecciona}
\small
\begin{algorithm}[H]
\scriptsize
\KwIn{caja $\mathbf{x}$}
\KwOut{variable $i^*$}
linealizar $f$ en el centro de $\mathbf{x}$\;
resolver el LP asociado\;
\If{el LP falla}{
  usar \texttt{smearsum}\;
}
calcular $s_i \gets |D_i|\,\mathrm{wid}(x_i)$\;
\Return $\argmax_i s_i$\;
\end{algorithm}

\begin{alertblock}{Idea clave}
La linealización de Taylor guía la selección de la variable más prometedora.
\end{alertblock}
\end{frame}

\small
\begin{columns}[T,onlytextwidth]
\column{0.56\textwidth}
\begin{block}{Comparación global}

\begin{frame}{El sobrecoste de linearizar es pequeño}
\small
\begin{block}{Mensaje clave}
\begin{itemize}
  \item \texttt{lsmear} resuelve un LP extra con el mismo Jacobiano usado por \texttt{smearsum}.
  \item Ese costo es pequeño frente al filtrado del solver, que domina el tiempo total.
\end{itemize}
\end{block}

\begin{exampleblock}{Variantes y limitación}
\begin{itemize}
  \item \texttt{lsmear-LR} y \texttt{lsmear-LRI} usan relajación lineal y requieren modificar el contractor.
  \item \texttt{lsmear}, \texttt{lsmear-LI} y \texttt{lsmear-MG} evitan ese cambio con una linearización independiente.
\end{itemize}
\end{exampleblock}
\end{frame}

\begin{frame}{La principal ganancia viene de ponderar restricciones}
\small
\begin{block}{Idea central}
\begin{itemize}
\item Ponderar restricciones con \(\lambda^*\) supera tratarlas como igualmente importantes.
\item \texttt{lsmear-MG} es la variante más simple y eficaz.
\item La relajación lineal completa no compensa su mayor complejidad.
\end{itemize}
\end{block}

\begin{alertblock}{Mensaje clave}
\textbf{Detectar} restricciones activas no basta; hay que incorporarlas en la ponderación.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{lsmear} usa información dual para biseccionar}
\small
\begin{block}{Idea del método}
\begin{itemize}
  \item Usa multiplicadores de Lagrange óptimos de una linealización.
  \item Calcula el \texttt{smear} sobre una función tipo Lagrangiano.
  \item Ignora variables ligadas solo a restricciones lineales.
\end{itemize}
\end{block}

\begin{alertblock}{Mensaje clave}
La selección mejora al ponderar restricciones y variables con información dual e intervalar.
\end{alertblock}

\begin{itemize}
  \item Extiende la heurística a otros solvers intervalares y variantes híbridas.
  \item Incorpora errores de violación, como en \texttt{ViolationTransfer}.
\end{itemize}
\end{frame}
