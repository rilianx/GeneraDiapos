\begin{frame}{Elegir mal la variable multiplica nodos}
\small
\begin{block}{Contexto}
La optimización global continua se resuelve determinísticamente con \textit{branch and bound} por intervalos.
\end{block}

\begin{alertblock}{Problema}
La bisección depende de elegir bien la variable; una mala elección multiplica nodos y tiempo.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{lsmear} pondera restricciones con \alert{pesos}}
\small
\begin{block}{Estrategias clásicas}
  \begin{itemize}
    \item \texttt{round-robin}: no usa información del sistema.
    \item \texttt{largest-first}: elige la variable con mayor dominio.
    \item \texttt{smear-based}: agrega impactos de todas las restricciones.
  \end{itemize}
\end{block}

\begin{exampleblock}{Idea de \texttt{lsmear}}
\[
D_i=\sum_{j=1}^m \lambda_j^*\,\operatorname{wid}(x_i)\,\max\!\bigl(|J_{ji}|\bigr)
\]
La variable se elige por su impacto ponderado en la función de Lagrange.
\end{exampleblock}

\begin{alertblock}{Limitación común}
  Tratan todas las restricciones como igualmente importantes.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{lsmear} evita la intrusión de \texttt{ViolationTransfer}}
\small
\begin{block}{Idea común}
\begin{itemize}
  \item Ambos usan multipliers duales y una función Lagrangiana.
  \item Seleccionan la variable con mayor impacto estimado.
\end{itemize}
\end{block}

\begin{alertblock}{Limitación de \texttt{ViolationTransfer}}
\begin{itemize}
  \item Requiere reformular la relajación convexa.
  \item Modifica el contractor y la estrategia del solver.
\end{itemize}
\end{alertblock}

\begin{exampleblock}{\texttt{lsmear}: alternativa simple}
\begin{itemize}
  \item Usa una linealización de Taylor del problema original.
  \item Calcula el impacto sin tocar el contractor.
\end{itemize}
\end{exampleblock}
\end{frame}

\begin{frame}{\texttt{lsmear} pondera restricciones con multiplicadores óptimos}
\small
\begin{block}{Idea del método}
\[
g_j(\mathbf{x}) \approx g_j(\operatorname{mid}(\mathbf{x})) + \operatorname{mid}(J_{j\cdot})\,(\mathbf{x}-\operatorname{mid}(\mathbf{x}))
\]
La restricción se linealiza en el punto medio de la caja.
\[
\lambda^* \leftarrow \argmin \{\text{LP linealizado}\}
\]
El programa lineal estima multiplicadores duales para ponderar restricciones.
\end{block}

\begin{alertblock}{Mensaje clave}
Cada restricción contribuye según su importancia dual, no por igual.
\end{alertblock}

\begin{itemize}
  \item Se calcula \(\mathcal{L}\) con esos pesos sobre la aproximación lineal.
  \item \texttt{lsmear} usa \(|D_i|\,\operatorname{wid}(x_i)\) para elegir la variable.
  \item Si el LP falla o es no acotado, \texttt{lsmear} cae a \texttt{smearsum}.
\end{itemize}
\end{frame}

\begin{frame}{\texttt{lsmear} selecciona en dos fases}
\small
\begin{algorithm}[H]
\scriptsize
\KwIn{caja $\mathbf{x}\in\R^n$, Jacobiano intervalar $J$}
\KwOut{índice $i^*$}
Resolver el LP con la variable objetivo no acotada\;
\If{no hay solución óptima o el óptimo es no acotado}{
  \Return decisión de \texttt{smearsum}\;
}
\For{$i=1,\dots,n$}{
  $D_i \gets \lambda_i^*$\;
  \For{$j=1,\dots,m$}{
    $D_i \gets D_i + J_{ji}\lambda_{n+j}^*$\;
  }
  $s_i \gets \left|D_i\right|\,\operatorname{wid}(x_i)$\;
}
\Return $\argmax_i s_i$\;
\end{algorithm}
\end{frame}

\begin{frame}{La evaluación usa IbexOpt y COCONUT}
\begin{block}{Configuración experimental}
\small
\begin{itemize}
  \item Implementamos \texttt{lsmear} y variantes en \texttt{IbexOpt}, con \texttt{HC4}, \texttt{ACID(HC4)} y relajación lineal.
  \item La selección de nodos usa \texttt{FeasibleDiving}; la precisión fija es $\epsilon=10^{-6}$.
  \item Evaluamos 76 instancias de \texttt{COCONUT}, corridas cinco veces por estrategia.
  \item Reportamos el promedio tras eliminar el mejor y el peor tiempo por instancia.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{\texttt{lsmear} pesa restricciones con multipliers duales}
\small
\begin{block}{Idea}
\begin{itemize}
  \item \texttt{lsmear} estima pesos con una linealización de Taylor y \textit{multipliers} duales óptimos.
  \item Usa esa estimación para ponderar las restricciones en la función lagrangiana local.
\end{itemize}
\end{block}

\begin{alertblock}{Mensaje clave}
\begin{itemize}
  \item \texttt{lsmear} y \texttt{lsmear-MG} mejoran a \texttt{ViolationTransfer}.
  \item \texttt{lsmear-MG} es más simple de implementar: no modifica el \textit{contractor}.
  \item El costo del LP es pequeño frente al \textit{pruning} del \texttt{solver}.
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{lsmear} mejora el branching sin intrusión}
\small
\begin{block}{Idea central}
\texttt{lsmear} pondera las restricciones con los multiplicadores óptimos de una linealización y calcula un smear tipo Lagrangiano.
\end{block}

\begin{itemize}
  \item \texttt{lsmear} y \texttt{lsmear-MG} superan todas las estrategias clásicas de bisección.
  \item \texttt{lsmear-MG} usa las restricciones originales para medir el impacto de las variables.
  \item Frente a \texttt{lsmear-LRI}, mejora el cálculo del impacto tras obtener los valores duales.
  \item A diferencia de \texttt{ViolationTransfer}, no modifica el contractor ni la estrategia global.
  \item Futuro: incorporar errores de violación y evaluar variables ligadas solo a restricciones lineales.
\end{itemize}
\end{frame}
