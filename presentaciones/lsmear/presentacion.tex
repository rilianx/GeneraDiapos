\begin{frame}{\texttt{NCOP}: contexto y bisección}
\small
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \begin{block}{Contexto}
  \begin{itemize}
    \item Optimización global continua y determinista mediante \textit{branch and bound} intervalar.
    \item Problema: $f:\R^n\to\R$ y $g:\R^n\to\R^m$ sobre una caja.
  \end{itemize}
  \end{block}

  \column{0.48\textwidth}
  \begin{exampleblock}{Bisección}
  \begin{itemize}
    \item Divide la caja actual en dos subcajas.
    \item Muchas estrategias usan información local para elegir la variable.
    \item Una mala ordenación inicial puede degradar el rendimiento.
  \end{itemize}
  \end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{El criterio smear clásico ignora pesos de restricciones}
\small
\begin{itemize}
  \item \texttt{round-robin} no usa información del sistema.
  \item \texttt{largest-first} ignora la estructura de derivadas.
  \item \texttt{smearsum}, \texttt{smearmax} y \texttt{smearsumrel} tratan todas las restricciones por igual.
  \item \texttt{ViolationTransfer} pondera restricciones, pero exige relajación convexa intrusiva.
\end{itemize}
\end{frame}

\begin{frame}{El solver estándar alterna poda y bifurcación}
\small
\begin{block}{Estructura básica}
\begin{itemize}
  \item Selecciona una caja activa, la bisecciona y filtra cada subcaja.
  \item \textbf{Poda}: actualiza la cota inferior con \texttt{contractors} sobre $x_o$.
  \item \textbf{Upper bounding}: busca soluciones factibles y mejora $UB$ y $x_{UB}$.
\end{itemize}
\end{block}

\begin{alertblock}{Cuello de botella}
La calidad de la variable de bisección domina el tamaño del árbol.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{lsmear} pondera restricciones con duales lineales}
\small
\begin{block}{Idea del método}
\begin{itemize}
  \item Linealiza cada $g_j(x)$ con Taylor de primer orden alrededor del centro de la caja.
  \item Resuelve un problema lineal con cotas de caja mediante \texttt{simplex}.
  \item Usa \alert{multiplicadores duales óptimos} para ponderar el Lagrangiano aproximado.
\end{itemize}
\end{block}

\begin{alertblock}{Si el LP falla}
Si no hay solución óptima o el valor es no acotado, \texttt{lsmear} cae a \texttt{smearsum}.
\end{alertblock}

\end{frame}

\begin{frame}{La variante básica usa dos fases}
\small
\begin{block}{Idea del método}
\begin{itemize}
  \item Fase 1: linealiza cada restricción alrededor del punto medio de la caja.
  \item Fase 2: mide el impacto variable por variable en el Lagrangiano.
  \item Los multiplicadores de la relajación lineal reemplazan a los del problema original.
\end{itemize}
\end{block}

\[
g_j(x)\approx g_j(\operatorname{mid}(\mathbf{x}))+\mathbf{J}_{j\cdot}(\operatorname{mid}(\mathbf{x}))^\top\!\left(x-\operatorname{mid}(\mathbf{x})\right)
\]
\small La restricción se aproxima con el término lineal de Taylor en el centro de la caja.

\[
L(x,\lambda^*)=f(x)+\sum_{j=1}^m \lambda_j^*\, g_j(x)
\]
\small Los pesos $\lambda^*$ provienen del problema lineal y estiman la importancia de cada restricción.

\end{frame}

\begin{frame}{La selección maximiza un smear ponderado}
\small
\begin{block}{Idea central}
\[
D_i=\lambda_i^*+\sum_{j=1}^{m}\lambda_{n+j}^*\,\mathbf{J}_{ji}
\]
\[
\text{impacto}(x_i)=\left|D_i\right|\,\mathrm{wid}(x_i)
\]
\end{block}
\small
\begin{itemize}
  \item \texttt{lsmear} usa multiplicadores duales óptimos para ponderar restricciones.
  \item \(\mathbf{J}_{ji}\) aproxima \(\partial g_j/\partial x_i\) con derivadas intervalares.
  \item El impacto combina sensibilidad y ancho del intervalo de \(x_i\).
  \item Se elige la variable con \alert{mayor magnitud de impacto}.
\end{itemize}
\begin{alertblock}{Limitación y fallback}
Si el programa lineal no tiene solución o es no acotado, se usa \texttt{smearsum}.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{lsmear} en dos fases}
\small
\begin{block}{Idea}
La estrategia usa una \alert{estimación de multiplicadores duales óptimos} para ponderar las restricciones.
\end{block}

\begin{algorithm}[H]
\scriptsize
\KwIn{caja $\mathbf{x}$, Jacobiano intervalar $\mathbf{J}$, restricciones $g$}
\KwOut{variable a biseccionar}
linealizar $g$ en torno a $\mathrm{mid}(\mathbf{x})$\;
resolver el LP con la cota del objetivo no acotada\;
\If{el LP no tiene solución o es no acotado}{
  \Return \texttt{smearsum}$(\mathbf{x},\mathbf{J})$\;
}
\For{$i \in 1,\dots,n$}{
  $D \gets \lambda_i^*$\;
  \For{$j \in 1,\dots,m$}{
    $D \gets D + \lambda_{n+j}^*\,\mathbf{J}_{ji}$\;
  }
  evaluar $\lvert D \cdot \mathrm{wid}(x_i)\rvert$\;
}
\Return la variable con mayor impacto\;
\end{algorithm}

\begin{alertblock}{Implementación}
\small
Se usa \texttt{SoPlex} para resolver el problema primal y el dual.
\end{alertblock}
\end{frame}

\begin{frame}{Cuatro variantes aíslan el efecto de cada decisión}
\small
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Linealización}
\begin{itemize}
  \item \texttt{lsmear-MG} usa el gradiente en el punto medio.
  \item \texttt{lsmear-LI} reemplaza \texttt{J\textsubscript{ji}} por su media en el lagrangiano.
  \item Ambas conservan una estimación local y barata.
\end{itemize}

\column{0.48\textwidth}
\textbf{Relajación}
\begin{itemize}
  \item \texttt{lsmear-LR} obtiene $\lambda^*$ desde una relajación lineal.
  \item \texttt{lsmear-LRI} combina esa relajación con el impacto lineal.
  \item \alert{Aíslan el efecto de la linealización y de los multiplicadores.}
\end{itemize}
\end{columns}

\end{frame}

\begin{frame}{\texttt{lsmear} supera las estrategias clásicas}
\small

\begin{block}{Contexto y problema}
\begin{itemize}
  \item \texttt{lsmear} es una nueva estrategia de selección de variables para \texttt{interval-based solvers}.
  \item Pondera restricciones con multiplicadores de Lagrange óptimos.
\end{itemize}
\end{block}

\begin{exampleblock}{Idea del método}
\begin{itemize}
  \item La \alert{linealización} del problema calcula los multiplicadores con el método simplex.
  \item \texttt{lsmear} define una función tipo Lagrangiano y mide su valor de smear.
  \item \texttt{lsmear-MG} calcula los impactos con las restricciones originales.
\end{itemize}
\end{exampleblock}

\begin{alertblock}{Resultado principal y trabajo futuro}
\begin{itemize}
  \item \texttt{lsmear} y \texttt{lsmear-MG} superan significativamente a las estrategias clásicas de bisección.
  \item \texttt{lsmear} es independiente del contractor basado en relajación convexa.
  \item Futuro: incorporar errores de violación para ponderar variables relacionadas.
\end{itemize}
\end{alertblock}
